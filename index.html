/************************************************* * 下貨點收回報系統 * Google Apps Script 後端 * * 工作表： * 大車 * A = ID * B = 大車名稱 * C = 啟用 * D = 固定置頂 * * 下貨人員 * A = ID * B = 姓名 * C = 啟用 * * 回報紀錄 * A = 日期 * B = 時間 * C = 員工 * D = 大車 * E = 下貨人員 * F = 件數 * G = 備註 * H = 照片資料夾 *************************************************/ /* ================================================= 基本設定 ================================================= */ const DRIVE_FOLDER_ID = "1b6WgCADGun85eMVlL1px2JTL0nkVyMUN"; const TRUCK_SHEET = "大車"; const STAFF_SHEET = "下貨人員"; const REPORT_SHEET = "回報紀錄"; // 每天早上 06:00 才算新的一天 const DAY_START_HOUR = 6; /* ================================================= GET ================================================= */ function doGet(e) { try { const data = getSystemData(); return ContentService .createTextOutput( JSON.stringify({ success: true, data: data }) ) .setMimeType( ContentService.MimeType.JSON ); } catch (error) { return ContentService .createTextOutput( JSON.stringify({ success: false, error: error.message }) ) .setMimeType( ContentService.MimeType.JSON ); } } /* ================================================= POST ================================================= */ function doPost(e) { try { if ( !e || !e.postData || !e.postData.contents ) { throw new Error( "沒有收到資料" ); } const payload = JSON.parse( e.postData.contents ); const result = saveReport( payload ); return ContentService .createTextOutput( JSON.stringify({ success: true, message: "回報成功", data: result }) ) .setMimeType( ContentService.MimeType.JSON ); } catch (error) { console.error( error ); return ContentService .createTextOutput( JSON.stringify({ success: false, error: error.message }) ) .setMimeType( ContentService.MimeType.JSON ); } } /* ================================================= 取得所有系統資料 ================================================= */ function getSystemData() { const ss = SpreadsheetApp .getActiveSpreadsheet(); if (!ss) { throw new Error( "找不到 Google 試算表" ); } /* ------------------------- 大車 ------------------------- */ const truckSheet = ss.getSheetByName( TRUCK_SHEET ); if (!truckSheet) { throw new Error( "找不到「大車」工作表" ); } const trucks = readTruckList( truckSheet ); /* ------------------------- 下貨人員 ------------------------- */ const staffSheet = ss.getSheetByName( STAFF_SHEET ); if (!staffSheet) { throw new Error( "找不到「下貨人員」工作表" ); } const staff = readEnabledList( staffSheet ); /* ------------------------- 今天大車最後下貨資訊 ------------------------- */ const truckLastUsed = getTruckLastUsedToday(); return { trucks: trucks, staff: staff, truckLastUsed: truckLastUsed, businessDate: getBusinessDateString() }; } /* ================================================= 讀取大車 A = ID B = 大車名稱 C = 啟用 D = 固定置頂 ================================================= */ function readTruckList(sheet) { const lastRow = sheet.getLastRow(); if ( lastRow < 2 ) { return []; } const values = sheet.getRange( 2, 1, lastRow - 1, 4 ).getValues(); const result = []; values.forEach( function(row) { const id = String( row[0] || "" ).trim(); const name = String( row[1] || "" ).trim(); const enabled = isTrueValue( row[2] ); const fixed = isTrueValue( row[3] ); if (!name) { return; } if (!enabled) { return; } result.push({ id: id, name: name, fixed: fixed }); } ); return result; } /* ================================================= 讀取下貨人員 A = ID B = 姓名 C = 啟用 ================================================= */ function readEnabledList(sheet) { const lastRow = sheet.getLastRow(); if ( lastRow < 2 ) { return []; } const values = sheet.getRange( 2, 1, lastRow - 1, 3 ).getValues(); const result = []; values.forEach( function(row) { const id = String( row[0] || "" ).trim(); const name = String( row[1] || "" ).trim(); const enabled = isTrueValue( row[2] ); if (!name) { return; } if (!enabled) { return; } result.push({ id: id, name: name }); } ); return result; } /* ================================================= TRUE 判斷 ================================================= */ function isTrueValue(value) { if ( value === true ) { return true; } const text = String( value || "" ) .trim() .toUpperCase(); return ( text === "TRUE" || text === "1" || text === "Y" || text === "YES" || text === "是" || text === "啟用" ); } /* ================================================= 儲存回報 ================================================= */ function saveReport(payload) { if (!payload) { throw new Error( "沒有收到回報資料" ); } const ss = SpreadsheetApp .getActiveSpreadsheet(); const reportSheet = ss.getSheetByName( REPORT_SHEET ); if (!reportSheet) { throw new Error( "找不到「回報紀錄」工作表" ); } const employee = String( payload.user || "" ).trim(); const trucks = Array.isArray( payload.trucks ) ? payload.trucks : []; const staff = Array.isArray( payload.staff ) ? payload.staff : []; const qtyStatus = String( payload.qtyStatus || "" ).trim(); const note = String( payload.note || "" ).trim(); /* ------------------------- 基本檢查 ------------------------- */ if (!employee) { throw new Error( "找不到員工姓名" ); } if ( trucks.length === 0 ) { throw new Error( "至少需要選擇一台大車" ); } if ( staff.length === 0 ) { throw new Error( "至少需要選擇一位下貨人員" ); } if (!qtyStatus) { throw new Error( "缺少件數狀態" ); } /* ------------------------- 時間 ------------------------- */ const now = new Date(); const timezone = Session.getScriptTimeZone() || "Asia/Taipei"; const businessDate = getBusinessDateString( now ); const timeText = Utilities.formatDate( now, timezone, "HH:mm:ss" ); /* ------------------------- 建立照片資料夾 ------------------------- */ let photoFolderUrl = ""; if ( Array.isArray( payload.photos ) && payload.photos.length > 0 ) { photoFolderUrl = savePhotosToFolder( payload.photos, employee, businessDate, trucks ); } /* ------------------------- 寫入試算表 ------------------------- */ const rowNumber = reportSheet.getLastRow() + 1; reportSheet .getRange( rowNumber, 1, 1, 8 ) .setValues([ [ businessDate, timeText, employee, trucks.join("、"), staff.join("、"), qtyStatus, note, photoFolderUrl ] ]); /* ------------------------- 照片欄位做成可點擊連結 ------------------------- */ if ( photoFolderUrl ) { reportSheet .getRange( rowNumber, 8 ) .setFormula( '=HYPERLINK("' + photoFolderUrl + '","查看照片")' ); } return { date: businessDate, time: timeText, employee: employee, trucks: trucks, staff: staff, photos: photoFolderUrl }; } /* ================================================= 建立照片資料夾並儲存照片 結構： 主資料夾 └── 2026-08-13 └── 大車名稱 └── 01_時間_員工.jpg └── 02_時間_員工.jpg 如果一次選多台大車： └── 2026-08-13 └── 大車1_大車2 ├── 照片1 ├── 照片2 ================================================= */ function savePhotosToFolder( photos, employee, dateText, trucks ) { const rootFolder = DriveApp.getFolderById( DRIVE_FOLDER_ID ); /* ------------------------- 日期資料夾 ------------------------- */ const dateFolder = getOrCreateFolder( rootFolder, dateText ); /* ------------------------- 大車資料夾 ------------------------- */ const truckFolderName = trucks .join("、") .replace( /[\\\/:*?"<>|]/g, "" ) .substring( 0, 100 ); const truckFolder = getOrCreateFolder( dateFolder, truckFolderName ); /* ------------------------- 時間 ------------------------- */ const timezone = Session.getScriptTimeZone() || "Asia/Taipei"; const timestamp = Utilities.formatDate( new Date(), timezone, "HHmmss" ); const safeEmployee = employee .replace( /[\\\/:*?"<>|]/g, "" ) .substring( 0, 30 ); const savedFiles = []; /* ------------------------- 儲存所有照片 ------------------------- */ photos.forEach( function( dataUrl, index ) { if (!dataUrl) { return; } try { const match = dataUrl.match( /^data:(image\/[^;]+);base64,(.+)$/ ); if (!match) { return; } const contentType = match[1]; const base64Data = match[2]; const bytes = Utilities.base64Decode( base64Data ); const blob = Utilities.newBlob( bytes, contentType ); const extension = contentType === "image/png" ? "png" : "jpg"; const fileName = String(index + 1) .padStart( 2, "0" ) + "_" + timestamp + "_" + safeEmployee + "." + extension; blob.setName( fileName ); const file = truckFolder.createFile( blob ); savedFiles.push( file ); } catch (error) { console.error( "照片儲存失敗：", error.message ); } } ); if ( savedFiles.length === 0 ) { return ""; } /* ------------------------- 建立這一次回報的資料夾 讓同一批照片更好找 ------------------------- */ const reportFolderName = timestamp + "_" + safeEmployee; const reportFolder = truckFolder.createFolder( reportFolderName ); /* ------------------------- 將照片移到這次回報資料夾 ------------------------- */ savedFiles.forEach( function(file) { file.moveTo( reportFolder ); } ); return reportFolder.getUrl(); } /* ================================================= 建立 / 取得資料夾 ================================================= */ function getOrCreateFolder( parentFolder, folderName ) { const folders = parentFolder.getFoldersByName( folderName ); if ( folders.hasNext() ) { return folders.next(); } return parentFolder.createFolder( folderName ); } /* ================================================= 取得今天各大車最後下貨資訊 回傳： { "3號大車": { time: "12:30:15", staff: "小明、小王" } } ================================================= */ function getTruckLastUsedToday() { const ss = SpreadsheetApp .getActiveSpreadsheet(); const sheet = ss.getSheetByName( REPORT_SHEET ); if (!sheet) { return {}; } const lastRow = sheet.getLastRow(); if ( lastRow < 2 ) { return {}; } const timezone = Session.getScriptTimeZone() || "Asia/Taipei"; const businessDate = getBusinessDateString(); const values = sheet.getRange( 2, 1, lastRow - 1, 8 ).getValues(); const result = {}; values.forEach( function(row) { const dateValue = row[0]; if (!dateValue) { return; } const dateText = normalizeDate( dateValue, timezone ); if ( dateText !== businessDate ) { return; } const timeText = String( row[1] || "" ).trim(); const truckText = String( row[3] || "" ).trim(); const staffText = String( row[4] || "" ).trim(); if (!truckText) { return; } const truckNames = truckText .split("、") .map( function(name) { return name.trim(); } ) .filter(Boolean); truckNames.forEach( function(truckName) { result[truckName] = { time: timeText, staff: staffText }; } ); } ); return result; } /* ================================================= 工作日 06:00 前算前一天 ================================================= */ function getBusinessDateString( date ) { date = date || new Date(); const timezone = Session.getScriptTimeZone() || "Asia/Taipei"; const hourText = Utilities.formatDate( date, timezone, "H" ); const hour = Number( hourText ); const adjusted = new Date( date.getTime() ); if ( hour < DAY_START_HOUR ) { adjusted.setDate( adjusted.getDate() - 1 ); } return Utilities.formatDate( adjusted, timezone, "yyyy-MM-dd" ); } /* ================================================= 日期格式整理 ================================================= */ function normalizeDate( value, timezone ) { if ( value instanceof Date ) { return Utilities.formatDate( value, timezone, "yyyy-MM-dd" ); } const text = String( value || "" ).trim(); if (!text) { return ""; } if ( /^\d{4}-\d{2}-\d{2}$/.test( text ) ) { return text; } const date = new Date( text ); if ( isNaN( date.getTime() ) ) { return ""; } return Utilities.formatDate( date, timezone, "yyyy-MM-dd" ); } /* ================================================= 測試系統 ================================================= */ function testSystem() { const data = getSystemData();
