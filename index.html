/************************************************* * 下貨點收回報系統 * Google Apps Script 後端 * * 功能： * 1. 取得啟用中的大車 * 2. 取得啟用中的下貨人員 * 3. 大車 D 欄可設定固定置頂 * 4. 每天以 06:00 為新的一天 * 5. 顯示大車最後一次下貨人員 * 6. 照片自動分類： * * Google Drive * └── 下貨點收回報 * └── 2026-08-13 * ├── 1號大車 * ├── 2號大車 * └── 香菇大類 * *************************************************/ /* ================================================= 基本設定 ================================================= */ const DRIVE_FOLDER_ID = "1b6WgCADGun85eMVlL1px2JTL0nkVyMUN"; const TRUCK_SHEET = "大車"; const STAFF_SHEET = "下貨人員"; const REPORT_SHEET = "回報紀錄"; // 每天早上 06:00 才算新的一天 const DAY_START_HOUR = 6; // 台灣時區 const TIMEZONE = "Asia/Taipei"; /* ================================================= GET 前端取得： - 大車 - 下貨人員 - 大車最後下貨資訊 ================================================= */ function doGet(e) { try { const data = getSystemData(); return ContentService .createTextOutput( JSON.stringify({ success: true, data: data }) ) .setMimeType( ContentService.MimeType.JSON ); } catch (error) { return ContentService .createTextOutput( JSON.stringify({ success: false, error: error.message }) ) .setMimeType( ContentService.MimeType.JSON ); } } /* ================================================= POST 接收前端回報 ================================================= */ function doPost(e) { try { if ( !e || !e.postData || !e.postData.contents ) { throw new Error( "沒有收到資料" ); } const payload = JSON.parse( e.postData.contents ); const result = saveReport( payload ); return ContentService .createTextOutput( JSON.stringify({ success: true, message: "回報成功", data: result }) ) .setMimeType( ContentService.MimeType.JSON ); } catch (error) { console.error( error ); return ContentService .createTextOutput( JSON.stringify({ success: false, error: error.message }) ) .setMimeType( ContentService.MimeType.JSON ); } } /* ================================================= 取得所有系統資料 ================================================= */ function getSystemData() { const ss = SpreadsheetApp .getActiveSpreadsheet(); if (!ss) { throw new Error( "找不到 Google 試算表" ); } /* ------------------------- 大車 ------------------------- */ const truckSheet = ss.getSheetByName( TRUCK_SHEET ); if (!truckSheet) { throw new Error( "找不到「大車」工作表" ); } const trucks = readTruckList( truckSheet ); /* ------------------------- 下貨人員 ------------------------- */ const staffSheet = ss.getSheetByName( STAFF_SHEET ); if (!staffSheet) { throw new Error( "找不到「下貨人員」工作表" ); } const staff = readEnabledList( staffSheet ); /* ------------------------- 今天各大車最後下貨資訊 ------------------------- */ const truckLastUsed = getTruckLastUsedToday(); return { trucks: trucks, staff: staff, truckLastUsed: truckLastUsed, businessDate: getBusinessDateString() }; } /* ================================================= 讀取大車 A = ID B = 大車名稱 C = 啟用 D = 固定置頂 ================================================= */ function readTruckList(sheet) { const lastRow = sheet.getLastRow(); if ( lastRow < 2 ) { return []; } const values = sheet.getRange( 2, 1, lastRow - 1, 4 ).getValues(); const result = []; values.forEach( function(row) { const id = String( row[0] || "" ).trim(); const name = String( row[1] || "" ).trim(); const enabled = isTrueValue( row[2] ); const fixed = isTrueValue( row[3] ); if (!name) { return; } if (!enabled) { return; } result.push({ id: id, name: name, fixed: fixed }); } ); return result; } /* ================================================= 讀取下貨人員 A = ID B = 姓名 C = 啟用 ================================================= */ function readEnabledList(sheet) { const lastRow = sheet.getLastRow(); if ( lastRow < 2 ) { return []; } const values = sheet.getRange( 2, 1, lastRow - 1, 3 ).getValues(); const result = []; values.forEach( function(row) { const id = String( row[0] || "" ).trim(); const name = String( row[1] || "" ).trim(); const enabled = isTrueValue( row[2] ); if (!name) { return; } if (!enabled) { return; } result.push({ id: id, name: name }); } ); return result; } /* ================================================= TRUE 判斷 ================================================= */ function isTrueValue(value) { if ( value === true ) { return true; } const text = String( value || "" ) .trim() .toUpperCase(); return ( text === "TRUE" || text === "1" || text === "Y" || text === "YES" || text === "是" || text === "啟用" ); } /* ================================================= 儲存回報 ================================================= */ function saveReport(payload) { if (!payload) { throw new Error( "沒有收到回報資料" ); } const ss = SpreadsheetApp .getActiveSpreadsheet(); const reportSheet = ss.getSheetByName( REPORT_SHEET ); if (!reportSheet) { throw new Error( "找不到「回報紀錄」工作表" ); } const employee = String( payload.user || "" ).trim(); const trucks = Array.isArray( payload.trucks ) ? payload.trucks : []; const staff = Array.isArray( payload.staff ) ? payload.staff : []; const qtyStatus = String( payload.qtyStatus || "" ).trim(); const note = String( payload.note || "" ).trim(); /* ------------------------- 基本檢查 ------------------------- */ if (!employee) { throw new Error( "找不到員工姓名" ); } if ( trucks.length === 0 ) { throw new Error( "至少需要選擇一台大車" ); } if ( staff.length === 0 ) { throw new Error( "至少需要選擇一位下貨人員" ); } if (!qtyStatus) { throw new Error( "缺少件數狀態" ); } /* ------------------------- 時間 ------------------------- */ const now = new Date(); const businessDate = getBusinessDateString( now ); const timeText = Utilities.formatDate( now, TIMEZONE, "HH:mm:ss" ); /* ------------------------- Google Drive 日期 → 大車 ------------------------- */ const rootFolder = DriveApp.getFolderById( DRIVE_FOLDER_ID ); const dateFolder = getOrCreateFolder( rootFolder, businessDate ); /* ------------------------- 儲存照片 ------------------------- */ const photoUrls = []; if ( Array.isArray( payload.photos ) ) { payload.photos.forEach( function( dataUrl, index ) { if (!dataUrl) { return; } /* * 如果一次選多台大車， * 同一組照片會分別放入 * 每一台大車的資料夾。 */ trucks.forEach( function(truckName) { const truckFolder = getOrCreateFolder( dateFolder, sanitizeFolderName( truckName ) ); const fileUrl = saveBase64Image( dataUrl, truckFolder, employee, businessDate, index + 1 ); if (fileUrl) { photoUrls.push( fileUrl ); } } ); } ); } /* ------------------------- 寫入試算表 ------------------------- */ reportSheet.appendRow([ businessDate, timeText, employee, trucks.join("、"), staff.join("、"), qtyStatus, note, photoUrls.join("\n") ]); return { date: businessDate, time: timeText, employee: employee, trucks: trucks, staff: staff, photos: photoUrls }; } /* ================================================= 建立／取得資料夾 ================================================= */ function getOrCreateFolder( parentFolder, folderName ) { const folders = parentFolder.getFoldersByName( folderName ); if ( folders.hasNext() ) { return folders.next(); } return parentFolder.createFolder( folderName ); } /* ================================================= 清理資料夾名稱 ================================================= */ function sanitizeFolderName( name ) { return String( name || "" ) .replace( /[\\\/:*?"<>|]/g, "" ) .trim() .substring( 0, 80 ); } /* ================================================= 儲存圖片 ================================================= */ function saveBase64Image( dataUrl, folder, employee, dateText, index ) { try { const match = dataUrl.match( /^data:(image\/[^;]+);base64,(.+)$/ ); if (!match) { return ""; } const contentType = match[1]; const base64Data = match[2]; const bytes = Utilities.base64Decode( base64Data ); const blob = Utilities.newBlob( bytes, contentType ); const extension = contentType === "image/png" ? "png" : "jpg"; const timestamp = Utilities.formatDate( new Date(), TIMEZONE, "HHmmss" ); const safeEmployee = employee .replace( /[\\\/:*?"<>|]/g, "" ) .substring( 0, 30 ); const fileName = dateText + "_" + timestamp + "_" + safeEmployee + "_" + index + "." + extension; blob.setName( fileName ); const file = folder.createFile( blob ); return file.getUrl(); } catch (error) { console.error( "圖片儲存失敗：", error.message ); return ""; } } /* ================================================= 取得今天各大車最後下貨資訊 * * 回傳： * * { * "3號大車": { * time: "10:32:15", * staff: ["阿賢","阿益"] * } * } ================================================= */ function getTruckLastUsedToday() { const ss = SpreadsheetApp .getActiveSpreadsheet(); const sheet = ss.getSheetByName( REPORT_SHEET ); if (!sheet) { return {}; } const lastRow = sheet.getLastRow(); if ( lastRow < 2 ) { return {}; } const businessDate = getBusinessDateString(); const values = sheet.getRange( 2, 1, lastRow - 1, 8 ).getValues(); const result = {}; values.forEach( function(row) { const dateValue = row[0]; if (!dateValue) { return; } const dateText = normalizeDate( dateValue ); if ( dateText !== businessDate ) { return; } const timeText = String( row[1] || "" ).trim(); const truckText = String( row[3] || "" ).trim(); const staffText = String( row[4] || "" ).trim(); if (!truckText) { return; } const truckNames = truckText .split("、") .map( function(name) { return name.trim(); } ) .filter(Boolean); const staffNames = staffText .split("、") .map( function(name) { return name.trim(); } ) .filter(Boolean); truckNames.forEach( function(truckName) { /* * 後面的紀錄會覆蓋前面的紀錄， * 因此得到該車最後一次回報。 */ result[truckName] = { time: timeText, staff: staffNames }; } ); } ); return result; } /* ================================================= 工作日 06:00 前算前一天 ================================================= */ function getBusinessDateString( date ) { date = date || new Date(); const dateText = Utilities.formatDate( date, TIMEZONE, "yyyy-MM-dd" ); const hour = Number( Utilities.formatDate( date, TIMEZONE, "H" ) ); if ( hour >= DAY_START_HOUR ) { return dateText; } const previousDay = new Date( date.getTime() - 24 * 60 * 60 * 1000 ); return Utilities.formatDate( previousDay, TIMEZONE, "yyyy-MM-dd" ); } /* ================================================= 日期格式整理 ================================================= */ function normalizeDate( value ) { if ( value instanceof Date ) { return Utilities.formatDate( value, TIMEZONE, "yyyy-MM-dd" ); } const text = String( value || "" ).trim(); if (!text) { return ""; } if ( /^\d{4}-\d{2}-\d{2}$/.test( text ) ) { return text; } const date = new Date( text ); if ( isNaN( date.getTime() ) ) { return ""; } return Utilities.formatDate( date, TIMEZONE, "yyyy-MM-dd" ); } /* ================================================= 測試系統 ================================================= */ function testSystem() { const data = getSystemData(); Logger.log( JSON.stringify( data, null, 2 ) ); }
